cmake_minimum_required(VERSION 3.20)
project(clock_theme_ui_test LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(CLOCK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CLOCK_ROOT}/managed_components/lvgl__lvgl" lvgl)
file(GLOB CLOCK_ASSETS "${CLOCK_ROOT}/main/generated/*.c")
add_executable(theme_ui smoke.cpp "${CLOCK_ROOT}/main/theme_view.cpp" ${CLOCK_ASSETS})
target_include_directories(theme_ui PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/stubs" "${CLOCK_ROOT}/main")
target_link_libraries(theme_ui PRIVATE lvgl)
